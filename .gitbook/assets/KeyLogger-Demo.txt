  <p class="muted">
    This page demonstrates how <strong>DOM input events</strong> can be observed by any script running on the page.
    It does <strong>not</strong> send data anywhere and requires you to click <em>Start demo</em> first.
    Passwords are masked in the display output.
  </p>

  <div class="grid">
    <section>

      <label for="demo-password">Password (masked in log)</label>
      <input id="demo-password" name="password" type="password" autocomplete="off" placeholder="Type a password..." />

      <label for="demo-notes">Notes</label>
      <textarea id="demo-notes" name="notes" placeholder="Type some free text..."></textarea>

      <div class="controls">

  <details>
    <summary><strong>What this demonstrates</strong> (click to expand)</summary>
    <ul>
      <li>Any script running on a page can attach <code>input</code> / <code>keydown</code> listeners.</li>
      <li>The browser delivers what the user types to those listeners.</li>
      <li>Password fields are masked in the demo’s display (shows only length).</li>
    </ul>
  </details>

  <script>
    (function () {
      // State
      let attached = false;
      const BUFFER_LIMIT = 5000; // prevent unbounded growth in demo log

      // Elements
      const $u   = document.getElementById('demo-username');
      const $p   = document.getElementById('demo-password');
      const $n   = document.getElementById('demo-notes');
      const $log = document.getElementById('log');
      const $start = document.getElementById('start');
      const $stop  = document.getElementById('stop');
      const $clear = document.getElementById('clear');

      // Utilities
      function now() {
        return new Date().toISOString().replace('T',' ').replace('Z',' UTC');
      }
      function appendLog(line) {
        const sanitized = String(line).replace(/\r?\n/g, ' ');
        let current = $log.value;               // FIX: initialize first
        const prefix = current.length ? '' : '';
        current = current + prefix + sanitized + "\n";
        if (current.length > BUFFER_LIMIT) {
          current = current.slice(current.length - BUFFER_LIMIT);
        }
        $log.value = current;
        $log.scrollTop = $log.scrollHeight;
      }
      function elLabel(el) {
        if (!el) return '(unknown)';
        const id = el.id ? `#${el.id}` : '';
        const nm = el.name ? `[name="${el.name}"]` : '';
        return `<${el.tagName.toLowerCase()}${id}${nm}>`;
      }

      // Masking for password fields in display
      function maskIfPassword(el, value) {
        if (el && el.type === 'password') {
          return '•'.repeat(String(value || '').length);
        }
        return value;
      }

      // Event handlers (kept simple and local)
      function onInput(e) {
        const target = e.target;
        const valueShown = maskIfPassword(target, target.value);
        appendLog(`[${now()}] input on ${elLabel(target)} -> value: "${valueShown}"`);
        console.debug('input event (structured):', {
          target: elLabel(target),
          valuePreview: valueShown,
          isPassword: target.type === 'password'
        });
      }
      function onKeydown(e) {
        const target = e.target;
        appendLog(`[${now()}] keydown on ${elLabel(target)} -> key="${e.key}" code="${e.code}"`);
        console.debug('keydown event (structured):', {
          target: elLabel(target),
          key: e.key,
          code: e.code,
          ctrl: e.ctrlKey, shift: e.shiftKey, alt: e.altKey, meta: e.metaKey
        });
      }

      // Attach / detach
      function attach() {
        if (attached) return;
        [$u, $p, $n].forEach(el => {
          el.addEventListener('input', onInput, { passive: true });
          el.addEventListener('keydown', onKeydown, { passive: true });
        });
        attached = true;
        $start.disabled = true;
        $stop.disabled  = false;
        appendLog(`[${now()}] Demo started. Listeners attached.`);
      }
      function detach() {
        if (!attached) return;
        [$u, $p, $n].forEach(el => {
          el.removeEventListener('input', onInput);
          el.removeEventListener('keydown', onKeydown);
        });
        attached = false;
        $start.disabled = false;
        $stop.disabled  = true;
        appendLog(`[${now()}] Demo stopped. Listeners detached.`);
      }

      // Wire up controls
      $start.addEventListener('click', attach);
      $stop.addEventListener('click', detach);
      $clear.addEventListener('click', () => { $log.value = ''; });

      // Initial state
      appendLog(`[${now()}] Ready. Click "Start demo" and then type in the fields.`);
    })();
  </script>
